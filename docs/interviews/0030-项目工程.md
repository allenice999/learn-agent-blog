## 1. 如何在vite中配置达到axios和lodash两个依赖单独打包为一个文件？
Vite 7及以下使用manualChunks实现
```js
import { defineConfig } from 'vite'

export default defineConfig({
  build: {
    rollupOptions: {
      output: {
        manualChunks: {
          // 将 axios 和 lodash 单独打包成一个名为 'utils-vendor' 的文件
          'utils-vendor': ['axios', 'lodash']
        }
      }
    }
  }
})
```
Vite 8已将manualChunks移除，需要改用 rolldownOptions.output.codeSplitting.groups 或 advancedChunks 配置

```js
import { defineConfig } from 'vite'

export default defineConfig({
  build: {
    rolldownOptions: {
      output: {
        codeSplitting: {
          groups: [
            {
              name: 'utils-vendor',
              // 用正则匹配 axios 和 lodash 的模块路径
              test: /[\\/]node_modules[\\/](axios|lodash)[\\/]/
            }
          ]
        }
      }
    }
  }
})
```
advancedChunks 提供了更细的控制维度（优先级、体积限制、共享计数），适合有多个依赖需要分层管理的场景
```js
// vite.config.ts
import { defineConfig } from 'vite'

export default defineConfig({
  build: {
    rolldownOptions: {
      output: {
        advancedChunks: {
          groups: [
            // 框架核心：优先级最高，单独分包，缓存最稳定
            {
              name: 'react-vendor',
              test: /[\\/]node_modules[\\/](react|react-dom|scheduler)[\\/]/,
              priority: 30,
              minSize: 20000,
              maxSize: 200000,
            },
            // 路由库：优先级次之
            {
              name: 'router',
              test: /[\\/]node_modules[\\/](react-router|react-router-dom)[\\/]/,
              priority: 25,
            },
            // UI 库：仅当被 2 个以上模块共享时才提取
            {
              name: 'ui-vendor',
              test: /[\\/]node_modules[\\/]@radix-ui[\\/]/,
              priority: 20,
              minShareCount: 2,
            },
            // 兜底 vendor：优先级最低，用 maxSize 防止单个包过大
            {
              name: 'vendor',
              test: /[\\/]node_modules[\\/]/,
              priority: 5,
              maxSize: 500000, // 超过约 500KB 会自动拆成 vendor、vendor-1……
            },
          ],
        },
      },
    },
  },
})
```


## 2. 项目中有ABC三个子项目，BC两个子项目依赖A导出函数，怎么做项目结构设计以最大化复用代码？
pnpm monorepo，将A项目的方法导出为一个独立库包，BC通过包名引入

> 循环依赖怎么办？
> - 推荐方法：下沉公共逻辑到common
> - 依赖倒置：AB循环依赖，若方法确实属于 B，但 A 又必须调用它，让 A 依赖接口而非 B：
> ```js
> // packages/a/src/index.ts
> // A 定义接口，不 import B
> export interface Formatter { format(user): string }
> let formatter: Formatter
> export function setFormatter(f: Formatter) { formatter = f }  // 注入点
> 
> export function logUser(id) {
>   return formatter.format(getUser(id))   // 只依赖抽象
> }
> ```
> ```js
> // packages/b/src/index.ts
> import { setFormatter } from '@proj/a'
> setFormatter({ format: formatUser })     // B 启动时注入自己
> ```
> A 不再 import B，依赖方向只剩 B → A，环打破。
